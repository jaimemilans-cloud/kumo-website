        function toggleBriefingSim() {
            const btnText = document.getElementById('playText');
            const icon = document.getElementById('playIcon');
            const transcript = document.getElementById('briefingTranscript');
            const waveBars = document.querySelectorAll('.wave-bar');

            if (!isPlaying) {
                isPlaying = true;
                btnText.innerText = "Pause Briefing";
                icon.className = "fa-solid fa-pause text-xs";
                
                let step = 0;
                transcript.innerText = `"${transcripts[0]}"`;

                interval = setInterval(() => {
                    step = (step + 1) % transcripts.length;
                    transcript.innerText = `"${transcripts[step]}"`;

                    waveBars.forEach(bar => {
                        const randomHeight = Math.floor(Math.random() * 40) + 10;
                        bar.style.height = `${randomHeight}px`;
                    });
                }, 3000);
            } else {
                isPlaying = false;
                btnText.innerText = "Play Morning Briefing";
                icon.className = "fa-solid fa-play text-xs";
                clearInterval(interval);
                waveBars.forEach(bar => bar.style.height = '16px');
            }
        }
